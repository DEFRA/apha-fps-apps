CREATE PROCEDURE [dbo].[sp_createFPSTotals]
AS
INSERT INTO FPSYearTotals
SELECT DISTINCT
    tlkpProject.ParentProject,
    tlkpProject.Program,
    CASE WHEN qryTotalAdditionalCosts.TotalAdditionalCosts IS NULL THEN 0 ELSE qryTotalAdditionalCosts.TotalAdditionalCosts END AS TotalAdditionalCosts,
    CASE WHEN qryTotalAnimalCosts.TotalAnimalCosts IS NULL THEN 0 ELSE qryTotalAnimalCosts.TotalAnimalCosts END AS TotalAnimalCosts,
    CASE WHEN qryTotalStaffCosts.TotalStaffCosts IS NULL THEN 0 ELSE qryTotalStaffCosts.TotalStaffCosts END AS TotalStaffCosts,
    CASE WHEN qryTotalTestCosts.TotalTestCosts IS NULL THEN 0 ELSE qryTotalTestCosts.TotalTestCosts END AS TotalTestCosts,
    CASE WHEN qryTotalAdditionalCosts.TotalAdditionalCosts IS NULL THEN 0 ELSE qryTotalAdditionalCosts.TotalAdditionalCosts END +
    CASE WHEN qryTotalAnimalCosts.TotalAnimalCosts IS NULL THEN 0 ELSE qryTotalAnimalCosts.TotalAnimalCosts END +
    CASE WHEN qryTotalStaffCosts.TotalStaffCosts IS NULL THEN 0 ELSE qryTotalStaffCosts.TotalStaffCosts END +
    CASE WHEN qryTotalTestCosts.TotalTestCosts IS NULL THEN 0 ELSE qryTotalTestCosts.TotalTestCosts END AS TotalCosts,
    tlkpProject.CustIncome,
    tlkpProject.TransferIncome,
    tlkpProject.CustIncome + tlkpProject.TransferIncome AS TotalIncome,
    tlkpProject.Budget_CVL,
    tlkpProject.CustIncome + tlkpProject.TransferIncome - (
        CASE WHEN qryTotalAdditionalCosts.TotalAdditionalCosts IS NULL THEN 0 ELSE qryTotalAdditionalCosts.TotalAdditionalCosts END +
        CASE WHEN qryTotalAnimalCosts.TotalAnimalCosts IS NULL THEN 0 ELSE qryTotalAnimalCosts.TotalAnimalCosts END +
        CASE WHEN qryTotalStaffCosts.TotalStaffCosts IS NULL THEN 0 ELSE qryTotalStaffCosts.TotalStaffCosts END +
        CASE WHEN qryTotalTestCosts.TotalTestCosts IS NULL THEN 0 ELSE qryTotalTestCosts.TotalTestCosts END
    ) AS RequiredProfit,
    tlkpProject.Manager,
    tlkpProject.Customer,
    tlkpProject.ProjectStatus,
    CASE WHEN tlkpProject.PVSIncome IS NULL THEN 0 ELSE tlkpProject.PVSIncome END AS PVSIncome,
    CASE WHEN tlkpProject.PlanCaseworkDebit IS NULL THEN 0 ELSE tlkpProject.PlanCaseworkDebit END AS PlanCaseworkDebit,
    CASE WHEN qryTotalStaffCosts.TotalPayCosts IS NULL THEN 0 ELSE qryTotalStaffCosts.TotalPayCosts END AS TotalPayCosts
FROM (((tlkpProject
LEFT JOIN qryTotalAdditionalCosts ON tlkpProject.ParentProject = qryTotalAdditionalCosts.JobCode)
LEFT JOIN qryTotalAnimalCosts ON tlkpProject.ParentProject = qryTotalAnimalCosts.JobCode)
LEFT JOIN qryTotalStaffCosts ON tlkpProject.ParentProject = qryTotalStaffCosts.Jobcode)
LEFT JOIN qryTotalTestCosts ON tlkpProject.ParentProject = qryTotalTestCosts.JobCode

  • Starts from tlkpProject and left joins cost summary sources.
  • Calculates additional costs, animal costs, staff costs, test costs, total costs, income values, required profit, PVS income, casework debit, and pay costs.
  • Uses explicit null-to-zero handling through CASE WHEN ... IS NULL THEN 0.

CREATE PROCEDURE [dbo].[sp_LoadFromFPS]
AS
BEGIN
declare @cFPSVersion varchar(20)
declare @FPSYear int
declare @RC int
Declare @sql VARCHAR(50)

SET @FPSYear=DatePart(yyyy,Getdate())-1
SET @cFPSVersion='FPS'+cast(@FPSYear as varchar(4))

if (SELECT Count(*) FROM master.dbo.sysdatabases where name =@cFPSVersion)>0
begin
set @sql = @cFPSVersion +'.dbo.sp_deleteFPSTotals'
Exec @sql
set @sql = @cFPSVersion +'.dbo.sp_createFPSTotals'
Exec @sql
Exec sp_DeleteYearsFPSData @cFPSVersion, @FPSYear
Exec sp_AddYearsFPSData @cFPSVersion, @FPSYear
end

if DATEPART(month, GETDATE()) >4
begin
SET @FPSYear=DatePart(yyyy,Getdate())
SET @cFPSVersion='FPS'+cast(@FPSYear as varchar(4))

if (SELECT Count(*) FROM master.dbo.sysdatabases where name =@cFPSVersion)>0
begin
set @sql = @cFPSVersion +'.dbo.sp_deleteFPSTotals'
Exec @sql
set @sql = @cFPSVersion +'.dbo.sp_createFPSTotals'
Exec @sql
Exec sp_DeleteYearsFPSData @cFPSVersion, @FPSYear
Exec sp_AddYearsFPSData @cFPSVersion, @FPSYear
end
end
else
begin
SET @FPSYear=DatePart(yyyy,Getdate())
SET @cFPSVersion='FPS'+cast(@FPSYear as varchar(4))

if (SELECT Count(*) FROM Master.dbo.sysdatabases where name =@cFPSVersion)>0
begin
DELETE from MY_tlkpProject_all WHERE Year=@FPSYear
Exec sp_AddMY_tlkpProject_All @cFPSVersion, @FPSYear
end
end
END
Logic Explanation for Developers
  • This is the main MAB Archive load orchestration procedure.
  • It first calculates the previous calendar year and constructs the FPS source database name.
  • If that previous-year FPS database exists, it rebuilds FPS totals and then deletes/reloads archive data for that year.
  • It then checks the current month. If the current month is greater than April, it performs the same full process for the current year.
  • If the current month is January-April, it only refreshes MY_tlkpProject_all for the current year.
Developer Note: The Jan-Apr partial refresh path is business-critical. Do not convert it into a full current-year refresh.

// --- Context calculation (equivalent of the SET @FPSYear / DATEPART logic) ---

public MabArchiveExecutionContext BuildExecutionContext()
{
    var now = GetNow();  // honours MABARCHIVE_TEST_UTCNOW env var for deterministic testing
    var currentYear  = now.Year;
    var currentMonth = now.Month;
    var previousYear = currentYear - 1;
    var primaryYear  = currentMonth > 4 ? currentYear : previousYear;

    return new MabArchiveExecutionContext(
        CurrentYear:              currentYear,
        PreviousYear:             previousYear,
        CurrentMonth:             currentMonth,
        PrimaryYear:              primaryYear,
        IncludePartialRefreshYear: currentMonth <= 4
    );
}

// --- Main orchestration (equivalent of the IF/ELSE branching) ---

public async Task ExecuteAsync(
    Guid runId,
    MabArchiveExecutionContext context,
    ITransactionWrapper transactionWrapper,
    CancellationToken ct)
{
    // Always process previous year (mirrors the unconditional first block)
    await ExecuteFullYearCycleAsync(context.PreviousYear, transactionWrapper, ct);

    if (context.CurrentMonth > 4)
    {
        // May-Dec: full archive cycle for current year too
        await ExecuteFullYearCycleAsync(context.CurrentYear, transactionWrapper, ct);
    }
    else
    {
        // Jan-Apr: partial refresh only — same as baseline else-branch
        await _dataService.RefreshProjectAllOnlyAsync(context.CurrentYear, ct);
    }
}

// --- Full year cycle (called once per year — replaces the inner begin/end blocks) ---

private async Task ExecuteFullYearCycleAsync(int year, ITransactionWrapper tx, CancellationToken ct)
{
    // Replaces: IF (SELECT Count(*) FROM master.dbo.sysdatabases ...) > 0
    if (!await _dataService.IsYearAvailableAsync(year, ct))
    {
        _logger.LogWarning("Year {Year} not found in fps.tblyearmaster — skipping cycle.", year);
        return;
    }

    _executionYearContext.FpsYear = year;

    await _totalsService.RebuildSourceTotalsAsync(year, ct);  // sp_deleteFPSTotals + sp_createFPSTotals
    await _dataService.DeleteYearDataAsync(year, ct);         // sp_DeleteYearsFPSData
    await _dataService.LoadYearDataAsync(year, ct);           // sp_AddYearsFPSData
}
